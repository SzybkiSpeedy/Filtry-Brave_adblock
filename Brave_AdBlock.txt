! Title: Moja Niestandardowa Lista Blokująca
! Description: Maksymalna ochrona sieciowa: Blokowanie requestów, IP, fingerprintingu, reklam i trackerów
! Version: 5.0
! Expires: 1 days

! ==========================================
! 1. BLOKOWANIE I KONTROLA POPRZEZ FILTROWANIE REQUESTÓW (SIĆ)
! ==========================================
! Blokowanie wszystkich ukrytych requestów wysyłanych przez skrypty w tle (XHR / Fetch / Beacon)
$third-party,xmlhttprequest
$ping

! Blokowanie requestów próbujących załadować obce strony wewnątrz ukrytych ramek (iFrames)
$subdocument,third-party

! Blokowanie wstrzykiwania złośliwych skryptów bezpośrednio w kodzie HTML strony (Inline Scripts)
$third-party,inline-script,domain=~google.com|~github.com

! Blokowanie requestów żądających pobrania zewnętrznych zasobów webowych (WebSockets śledzące w tle)
$websocket,third-party

! Wykorzystanie nagłówków CSP do odcięcia requestów do nieznanych zewnętrznych skryptów i baz danych
$third-party,csp=script-src 'self' 'unsafe-inline' https://*.google.com https://*.github.com

! ==========================================
! 2. OCHRONA PRZED ROZPOZNAWANIEM URZĄDZENIA I PRZEGLĄDARKI (ANTI-FINGERPRINTING)
! ==========================================
! Blokowanie requestów i skryptów badających unikalność urządzenia (Canvas, Audio, WebGL)
$domain=~brave.com,fingerprint
||fingerprintjs.com^$important
||cdn.jsdelivr.net/npm/@fingerprintjs/$script
||openfpcdn.io^$important
||valve.github.io/fingerprintjs2/$script

! Blokowanie dostępu do informacji o sprzęcie (bateria, stan sieci, sensory, Bluetooth)
$third-party,domain=~google.com,csp=navigator.battery; navigator.connection; navigator.userAgent

! Zapobieganie requestom czytającym czcionki systemowe i rozdzielczość ekranu
$third-party,script,domain=~github.com|~google.com

! ==========================================

! 4. GLOBALNE BLOKOWANIE COOKIES I BANERÓW RODO/ZGÓD

! Ukrywanie elementów graficznych: okienek i zasłon cookies na stronach

##[id*="cookie" i], [class*="cookie" i], [class*="consent" i]

! ==========================================
! 5. CZYSZCZENIE I BLOKOWANIE ŚLEDZĄCYCH HIPERŁĄCZY (URL CLEANING)
! ==========================================
! Usuwanie parametrów śledzących z linków (np. Facebook, Google, TikTok)
$removeparam=fbclid
$removeparam=utm_source
$removeparam=utm_medium
$removeparam=utm_campaign
$removeparam=gclid
$removeparam=gclsrc
$removeparam=ttclid
$removeparam=twclid
$removeparam=_openstat

! Blokowanie hiperłączy przekierowujących przez serwery śledzące (Click-tracking)
$explicit,third-party,link

! ==========================================
! 6. BLOKOWANIE REKLAM OTWIERAJĄCYCH NOWE KARTY (POP-UP / POP-UNDER)
! ==========================================
! Zakaz otwierania nowych okien przez skrypty stron trzecich (np. przy próbie pobierania)

$popunder,third-party
||onclickads.net^$important
||popads.net^$important
||popcash.net^$important
